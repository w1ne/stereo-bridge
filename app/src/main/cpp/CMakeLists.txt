cmake_minimum_required(VERSION 3.22.1)
project(stereobridge C)
set(LC3_ROOT ${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/liblc3)
file(GLOB LC3_SOURCES ${LC3_ROOT}/src/*.c)
add_library(stereobridge_codec SHARED codec.c ${LC3_SOURCES})
target_include_directories(stereobridge_codec PRIVATE ${LC3_ROOT}/include ${LC3_ROOT}/src)
target_compile_definitions(stereobridge_codec PRIVATE LC3_PLUS=1 LC3_PLUS_HR=1)
target_link_libraries(stereobridge_codec m)
target_link_options(stereobridge_codec PRIVATE "-Wl,-z,max-page-size=16384")
